cmake_minimum_required(VERSION 3.16)
project(armor_detection LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui videoio)
find_package(Eigen3 REQUIRED)
add_library(armor_detector ../lessons/lesson03_detection.cpp ../lessons/lesson03_neural.cpp)
target_include_directories(armor_detector PUBLIC ../include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(armor_detector PUBLIC ${OpenCV_LIBS} Eigen3::Eigen)
target_compile_options(armor_detector PRIVATE -Wall -Wextra -Wpedantic)
add_executable(armor_video armor_video.cpp)
target_link_libraries(armor_video PRIVATE armor_detector)
add_executable(lesson03_contract tests/contract.cpp)
target_link_libraries(lesson03_contract PRIVATE armor_detector)
enable_testing()
add_test(NAME lesson03_contract COMMAND lesson03_contract)
add_executable(armor_images armor_images.cpp)
target_link_libraries(armor_images PRIVATE armor_detector)
