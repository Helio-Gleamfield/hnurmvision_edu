cmake_minimum_required(VERSION 3.16)
project(hnurmvision_training LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(hnurm_interfaces REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui calib3d videoio imgcodecs)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
add_library(training_io io/config.cpp)
target_include_directories(training_io PUBLIC include io ${OpenCV_INCLUDE_DIRS})
target_link_libraries(training_io Eigen3::Eigen yaml-cpp ${OpenCV_LIBS})
ament_target_dependencies(training_io rclcpp sensor_msgs hnurm_interfaces)
target_compile_options(training_io PRIVATE -Wall -Wextra -Wpedantic)
add_library(training_lessons lessons/lesson02_io.cpp
 lessons/lesson03_detection.cpp lessons/lesson03_neural.cpp lessons/lesson04_tracking.cpp
 lessons/lesson05_aiming.cpp)
target_link_libraries(training_lessons PUBLIC training_io)
target_compile_options(training_lessons PRIVATE -Wall -Wextra -Wpedantic)
add_executable(training src/main.cpp)
target_link_libraries(training PRIVATE training_lessons)

include(CTest)
if(BUILD_TESTING)
  add_executable(lesson02_contract_test tests/lesson02_contract_test.cpp)
  target_link_libraries(lesson02_contract_test PRIVATE training_lessons)
  add_test(NAME lesson02_contract COMMAND lesson02_contract_test)
  add_executable(lesson02_ros_test tests/lesson02_ros_test.cpp)
  target_link_libraries(lesson02_ros_test PRIVATE training_lessons)
  add_test(NAME lesson02_ros_inprocess COMMAND lesson02_ros_test)
  set_tests_properties(lesson02_ros_inprocess PROPERTIES TIMEOUT 30
    ENVIRONMENT "ROS_DOMAIN_ID=90;ROS_LOCALHOST_ONLY=1")
endif()

add_executable(armor_video lesson03/armor_video.cpp)
target_link_libraries(armor_video PRIVATE training_lessons)
add_executable(lesson03_contract_test lesson03/tests/contract.cpp)
target_link_libraries(lesson03_contract_test PRIVATE training_lessons)
add_test(NAME lesson03_contract COMMAND lesson03_contract_test)
add_executable(camera_record lesson03/camera_record.cpp)
target_link_libraries(camera_record PRIVATE training_lessons)
add_executable(armor_images lesson03/armor_images.cpp)
target_link_libraries(armor_images PRIVATE training_lessons)
add_executable(camera_replay lesson03/camera_replay.cpp)
target_link_libraries(camera_replay PRIVATE training_io)
